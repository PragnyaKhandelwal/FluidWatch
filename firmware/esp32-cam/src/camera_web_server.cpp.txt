// CAMERA TUNING + LIVE PREVIEW WEB SERVER - not part of the real firmware.
//
// Hosts a page on the ESP32-CAM itself with a live MJPEG stream and a control
// for EVERY camera setting in src/capture_config.cpp (same names, ranges and
// help text as the real firmware), plus the values the sensor reports it is
// actually using. Use it to find the best settings for the real bottle,
// mounting and lighting, then copy the generated `fleet.py config ...`
// command at the bottom of the page to apply them to the deployed firmware.
//
// HOW TO USE:
//   1. Rename src/main.cpp to src/main.cpp.bak (that's the real
//      firmware - swap back to it after you're done previewing).
//   2. Rename this file to src/main.cpp.
//   3. Flash it over serial (hold IO0 to GND, tap RST), release IO0, tap RST.
//   4. Watch the serial monitor for a line like:
//        [WebServer] open http://192.168.1.42/ in a browser
//      Open that URL from a device on the same WiFi network.
//
// The camera streams continuously here, so it runs warm - don't leave this
// sketch running unattended. The real firmware powers the camera only per photo.

// handler (on the web server's task) stops grabbing frames first.

httpd_handle_t controlServer = nullptr; // port 80: page, settings, snapshots
httpd_handle_t streamServer = nullptr;  // port 81: the never-ending MJPEG stream

// Streaming runs at full speed (as the original firmware did) unless cpu_mhz
// was changed on the page.
uint32_t streamCpuMhz() {
    return settings.isSet(P_CPU_MHZ) ? settings[P_CPU_MHZ] : 240;
}

esp_err_t sendJson(httpd_req_t *req, const JsonDocument &doc) {
    String body;
    serializeJson(doc, body);
    httpd_resp_set_type(req, "application/json");
    return httpd_resp_send(req, body.c_str(), body.length());
}

esp_err_t indexHandler(httpd_req_t *req) {
    httpd_resp_set_type(req, "text/html");
    return httpd_resp_send(req, INDEX_HTML, HTTPD_RESP_USE_STRLEN);
}

// Every setting with its range, default, help text and current value.
esp_err_t paramsHandler(httpd_req_t *req) {
    JsonDocument doc;
    doc["device"] = DEVICE_ID;
    JsonArray list = doc["params"].to<JsonArray>();
    for (int i = 0; i < P_COUNT; ++i) {
        JsonObject p = list.add<JsonObject>();
        p["key"] = kParams[i].key;
        p["min"] = kParams[i].min;
        p["max"] = kParams[i].max;
        p["def"] = kParams[i].def;
        p["help"] = kParams[i].help;
        p["value"] = settings[(Param)i];
    }
    return sendJson(req, doc);
}

// What the sensor driver reports it is using right now.
esp_err_t statusHandler(httpd_req_t *req) {
    JsonDocument doc;
    camera::report(doc.to<JsonObject>());
    doc["cpu_mhz"] = getCpuFrequencyMhz();
    float t = thermal::celsius();
    if (!isnan(t)) {
        doc["chip_temp_c"] = roundf(t * 10) / 10;
    }
    doc["thermal"] = thermal::stateName();
    doc["heap_free"] = ESP.getFreeHeap();
    return sendJson(req, doc);
}

esp_err_t thermalHandler(httpd_req_t *req) {
    JsonDocument doc;
    float t = thermal::celsius();
    if (!isnan(t)) {
        doc["chip_temp_c"] = roundf(t * 10) / 10;
    }
    doc["state"] = thermal::stateName();
    doc["cutoff_c"] = settings[P_TEMP_CUTOFF_C];
    return sendJson(req, doc);
}

esp_err_t captureHandler(httpd_req_t *req) {
    if (!cameraAvailable) {
        httpd_resp_set_status(req, "503 Service Unavailable");
        return httpd_resp_send(req, "camera paused - chip cooling down", HTTPD_RESP_USE_STRLEN);
    }
    camera_fb_t *fb = esp_camera_fb_get();
    if (fb == nullptr) {
        httpd_resp_send_500(req);
        return ESP_FAIL;
    }
    httpd_resp_set_type(req, "image/jpeg");
    httpd_resp_set_hdr(req, "Content-Disposition", "inline; filename=capture.jpg");
    esp_err_t res = httpd_resp_send(req, (const char *)fb->buf, fb->len);
    esp_camera_fb_return(fb);
    return res;
}

// --- MJPEG stream: standard multipart/x-mixed-replace technique -----------

constexpr const char *kStreamContentType = "multipart/x-mixed-replace;boundary=frame";
constexpr const char *kStreamBoundary = "\r\n--frame\r\n";
constexpr const char *kStreamPartHeader = "Content-Type: image/jpeg\r\nContent-Length: %u\r\n\r\n";

esp_err_t streamHandler(httpd_req_t *req) {
    esp_err_t res = httpd_resp_set_type(req, kStreamContentType);
    if (res != ESP_OK) {
        return res;
    }

    char partBuf[64];
    while (cameraAvailable) {
        camera_fb_t *fb = esp_camera_fb_get();
        if (fb == nullptr) {
            Serial.println("[Stream] capture failed");
            res = ESP_FAIL;
            break;
        }

        res = httpd_resp_send_chunk(req, kStreamBoundary, strlen(kStreamBoundary));
        if (res == ESP_OK) {
            size_t headerLen = snprintf(partBuf, sizeof(partBuf), kStreamPartHeader, fb->len);
            res = httpd_resp_send_chunk(req, partBuf, headerLen);
        }
        if (res == ESP_OK) {
            res = httpd_resp_send_chunk(req, (const char *)fb->buf, fb->len);
        }

        esp_camera_fb_return(fb);

        if (res != ESP_OK) {
            break; // client disconnected
        }
    }
    return res;
}

// --- live settings adjustment: /control?var=<key>&val=<int> ----------------

// Stops the stream handler (on the stream server's task) from grabbing
// frames, and gives it time to finish the one it is sending.
void pauseStream() {
    cameraAvailable = false;
    delay(600);
}

esp_err_t controlHandler(httpd_req_t *req) {
    char query[96];
    char varName[32];
    char valStr[16];

    if (httpd_req_get_url_query_str(req, query, sizeof(query)) != ESP_OK ||
        httpd_query_key_value(query, "var", varName, sizeof(varName)) != ESP_OK ||
        httpd_query_key_value(query, "val", valStr, sizeof(valStr)) != ESP_OK) {
        httpd_resp_send_404(req);
        return ESP_FAIL;
    }

    JsonDocument change;
    change[varName] = atoi(valStr);
    int32_t oldFrameSize = settings[P_FRAME_SIZE];
    int32_t oldXclk = settings[P_XCLK_MHZ];
    String errors;
    settings.apply(change.as<JsonObjectConst>(), errors);

    String result = errors.length() > 0 ? errors : String("OK");
    if (errors.length() == 0 && !cameraAvailable) {
        result = "saved - applies when the chip has cooled down";
    } else if (errors.length() == 0) {
        if (!strcmp(varName, "cpu_mhz")) {
            // A clock change mid-frame breaks the stream's transfer: pause it,
            // switch, resume. The page reconnects the stream afterwards.
            pauseStream();
            setCpuFrequencyMhz(settings[P_CPU_MHZ]);
            cameraAvailable = true;
        } else if (!strcmp(varName, "flash")) {
            digitalWrite(FLASH_LED_GPIO_NUM, settings[P_FLASH] ? HIGH : LOW); // live preview of the LED
        } else if (settings[P_FRAME_SIZE] != oldFrameSize || settings[P_XCLK_MHZ] != oldXclk) {
            pauseStream();
            camera::powerDown(); // frame buffers and XCLK are fixed at init
            if (camera::powerUp(settings)) {
                cameraAvailable = true;
            } else {
                result = "camera re-init failed";
            }
        } else {
            String unsupported;
            if (camera::applySettings(settings, unsupported) > 0) {
                result = "sensor rejected: " + unsupported;
            }
        }
    }

    httpd_resp_set_type(req, "text/plain");
    return httpd_resp_send(req, result.c_str(), result.length());
}

void startWebServers() {
    httpd_config_t config = HTTPD_DEFAULT_CONFIG();
    config.max_uri_handlers = 8;

    httpd_uri_t indexUri = {"/", HTTP_GET, indexHandler, nullptr};
    httpd_uri_t paramsUri = {"/params", HTTP_GET, paramsHandler, nullptr};
    httpd_uri_t statusUri = {"/status", HTTP_GET, statusHandler, nullptr};
    httpd_uri_t captureUri = {"/capture", HTTP_GET, captureHandler, nullptr};
    httpd_uri_t controlUri = {"/control", HTTP_GET, controlHandler, nullptr};
    httpd_uri_t thermalUri = {"/thermal", HTTP_GET, thermalHandler, nullptr};
    if (httpd_start(&controlServer, &config) == ESP_OK) {
        httpd_register_uri_handler(controlServer, &thermalUri);
        httpd_register_uri_handler(controlServer, &indexUri);
        httpd_register_uri_handler(controlServer, &paramsUri);
        httpd_register_uri_handler(controlServer, &statusUri);
        httpd_register_uri_handler(controlServer, &captureUri);
        httpd_register_uri_handler(controlServer, &controlUri);
    } else {
        Serial.println("[WebServer] failed to start control server");
    }

    // The stream handler never returns while a browser watches, so it gets
    // its own server - otherwise the settings controls would freeze.
    config.server_port = 81;
    config.ctrl_port += 1;
    httpd_uri_t streamUri = {"/stream", HTTP_GET, streamHandler, nullptr};
    if (httpd_start(&streamServer, &config) == ESP_OK) {
        httpd_register_uri_handler(streamServer, &streamUri);
    } else {
        Serial.println("[WebServer] failed to start stream server");
    }
}

} // namespace

void setup() {
    Serial.begin(115200);
    delay(300);
    Serial.println("\n[Camera Web Server] booting...");

    pinMode(FLASH_LED_GPIO_NUM, OUTPUT);
    digitalWrite(FLASH_LED_GPIO_NUM, LOW);
    setCpuFrequencyMhz(streamCpuMhz());

    if (!camera::powerUp(settings)) {
        Serial.println("[main] camera init failed, halting");
        while (true) {
            delay(1000);
        }
    }

    wifiManager.begin(WIFI_SSID, WIFI_PASSWORD);
    if (!wifiManager.isConnected()) {
        Serial.println("[main] WiFi not connected - can't serve a webpage without it, halting");
        while (true) {
            delay(1000);
        }
    }

    startWebServers();
    if (MDNS.begin("fluidwatch")) {
        MDNS.addService("http", "tcp", 80);
    }
}

// Streaming keeps the camera on, so this sketch runs the warmest: pause the
// camera at the cutoff and bring it back once the chip has cooled.
void handleThermal() {
    if (!thermal::update(settings[P_TEMP_CUTOFF_C])) {
        return;
    }
    Serial.printf("[Thermal] %s (chip %.1f C, cutoff %d C)\n", thermal::stateName(), thermal::celsius(),
                  (int)settings[P_TEMP_CUTOFF_C]);
    if (thermal::captureAllowed()) {
        setCpuFrequencyMhz(streamCpuMhz());
        if (camera::powerUp(settings)) {
            cameraAvailable = true;
        }
    } else {
        cameraAvailable = false;
        delay(600); // let the stream handler finish its current frame
        camera::powerDown();
        setCpuFrequencyMhz(80);
    }
}

void loop() {
    wifiManager.ensureConnected();
    handleThermal();
    // Repeated, so the address can be read whenever a serial monitor attaches.
    static unsigned long lastAnnounceMs = 0;
    if (millis() - lastAnnounceMs >= 10000UL) {
        lastAnnounceMs = millis();
        Serial.printf("[WebServer] open http://%s/ or http://fluidwatch.local/ - chip %.1f C (%s, raw %.2f)\n",
                      WiFi.localIP().toString().c_str(), thermal::celsius(), thermal::stateName(),
                      thermal::rawCelsius());
    }
    delay(200);
}
